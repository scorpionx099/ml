
import numpy as np
import random
import matplotlib.pyplot as plt

size = 4
states = size * size
actions = 4
goal = 15

Q = np.zeros((states, actions))

alpha = 0.1
gamma = 0.9
epsilon = 1.0
rewards = []

def next_state(state, action):
    row, col = divmod(state, size)

    if action == 0:
        row = max(0, row - 1)
    elif action == 1:
        row = min(size - 1, row + 1)
    elif action == 2:
        col = max(0, col - 1)
    else:
        col = min(size - 1, col + 1)

    return row * size + col

# Training
for episode in range(2000):
    state = 0
    total_reward = 0

    for step in range(50):
        if random.random() < epsilon:
            action = random.randint(0, 3)
        else:
            action = np.argmax(Q[state])

        new_state = next_state(state, action)
        reward = 10 if new_state == goal else -1

        Q[state, action] += alpha * (
            reward + gamma * np.max(Q[new_state]) - Q[state, action]
        )

        state = new_state
        total_reward += reward

        if state == goal:
            break

    rewards.append(total_reward)
    epsilon = max(0.01, epsilon * 0.995)

print("Training Completed!")
print("Final Epsilon:", epsilon)
print("Average Reward:", np.mean(rewards[-100:]))

print("\nQ-Table:")
print("       Up      Down    Left    Right")

for i in range(states):
    print(f"State {i:2d}:", np.round(Q[i], 2))

arrows = ["↑", "↓", "←", "→"]

print("\nOptimal Policy:")
print("S = Start, G = Goal")

for i in range(states):
    if i == 0:
        print(" S ", end=" ")
    elif i == goal:
        print(" G ", end=" ")
    else:
        print(f" {arrows[np.argmax(Q[i])]} ", end=" ")

    if (i + 1) % size == 0:
        print()

state = 0
path = [state]

for step in range(50):
    if state == goal:
        break

    action = np.argmax(Q[state])
    state = next_state(state, action)
    path.append(state)

print("\nPath followed:")

for i, state in enumerate(path):
    row, col = divmod(state, size)
    print(f"Step {i}: ({row}, {col})")

if state == goal:
    print("Goal Reached Successfully!")
else:
    print("Goal Not Reached!")

plt.plot(np.convolve(rewards, np.ones(100)/100, mode="valid"))
plt.xlabel("Episodes")
plt.ylabel("Average Reward")
plt.title("Q-Learning Training Performance")
plt.grid()
plt.show()
